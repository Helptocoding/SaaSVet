def dump(path, value):
    assert not path.exists(), path
    text = json.dumps(value, indent=2, default=str)+'\n'

class ServerObserver:
    def __init__(self):
        self.plans=[]; self.statements=[]; self.errors=[]; self.proc=None
    def consume(self, row):
        if not row: return
        timestamp,pid,db,body=row
        if 'ROW_NUMBER()' not in body or 'agent_conversation' not in body: return
        if 'EXPLAIN' in body.split('ROW_NUMBER()',1)[0]: return
        import re
        duration=re.search(r'duration: ([0-9.]+) ms',body)
        if not duration: return
        record={'log_timestamp':timestamp,'backend_pid':int(pid),'database':db,'server_ms':float(duration.group(1))}
        if 'plan:' in body:
            try:
                plan=json.loads(body[body.index('{'):])
                query=plan.get('Query Text','')
                if 'ROW_NUMBER()' in query and not query.startswith('EXPLAIN'):
                    record['explain_event']=plan
                    self.plans.append(record)
            except (ValueError,json.JSONDecodeError): self.errors.append({'pid':pid,'kind':'selector_plan_parse'})
        elif 'statement:' in body or 'execute ' in body:
            record['exact_sql']=body.split('statement:',1)[-1].strip()
            self.statements.append(record)
    def reader(self):
        previous=None
        for line in self.proc.stdout:
            if line.startswith('DIAG|'):
                self.consume(previous)
                previous=line.rstrip('\r\n').split('|',4)[1:]
            elif previous: previous[3]+=line
        self.consume(previous)
    def start(self):
        assert not DATA.exists()
        with socket.socket() as s: assert s.connect_ex(('127.0.0.1',PORT))!=0
        cmd(PG/'initdb.exe','-D',DATA,'-U','b9e','-A','trust','--encoding=UTF8','--locale=C')
        args=['rtk','proxy',str(PG/'postgres.exe'),'-D',str(DATA),'-p',str(PORT),'-h','127.0.0.1',
            '-c','fsync=on','-c','synchronous_commit=on','-c','logging_collector=off',
            '-c','log_destination=stderr','-c','log_line_prefix=DIAG|%m|%p|%d| ',
            '-c','log_timezone=UTC','-c','log_parameter_max_length=0','-c','log_parameter_max_length_on_error=0',
            '-c','session_preload_libraries=auto_explain','-c','auto_explain.log_min_duration=0',
            '-c','auto_explain.log_analyze=on','-c','auto_explain.log_buffers=on',
            '-c','auto_explain.log_timing=on','-c','auto_explain.log_format=json',
            '-c','auto_explain.log_verbose=on','-c','log_min_duration_statement=0']
        self.proc=subprocess.Popen(args,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,
            text=True,encoding='utf-8',errors='replace',creationflags=subprocess.CREATE_NO_WINDOW)
        self.thread=threading.Thread(target=self.reader); self.thread.start()
        for retry in range(150):
            try:

                    row=c.execute("SELECT version(),current_setting('data_directory'),current_setting('fsync'),current_setting('synchronous_commit')").fetchone()
                    assert '16.15' in row[0] and Path(row[1]).resolve()==DATA.resolve() and row[2:]==('on','on')
                    c.execute('CREATE DATABASE '+TEMPLATE)
                return row
            except __import__('psycopg').OperationalError: time.sleep(.2)
        raise RuntimeError('STOP owned isolated PostgreSQL unavailable')
    def close(self):
        from django.db import connections
        connections.close_all()
        with connect(autocommit=True) as c:
            clients=c.execute("SELECT pid,datname FROM pg_stat_activity WHERE backend_type='client backend' AND pid<>pg_backend_pid()").fetchall()
            locks=c.execute('SELECT count(*) FROM pg_locks WHERE NOT granted').fetchone()[0]

            'historical':False,'raw_nonselector_sql_persisted':False,'contemporaneous_diagnostic_plans':True})
        return {'clients':0,'waiting_locks':0,'prepared':0,'dropped':names,'port':PORT,'port_closed':True,

class SQLTrace:
    def __init__(self): self.rows=[]
    def __call__(self,execute,query,params,many,context):
        start_ns=time.perf_counter_ns(); wall=time.time_ns()
        try: return execute(query,params,many,context)
        finally: self.rows.append({'sql':query,'params':params,'wall_start_ns':wall,
            'wall_end_ns':time.time_ns(),'elapsed_ns':time.perf_counter_ns()-start_ns})

def full_rows(name):
    from psycopg import sql
    result={}
    with connect(name,autocommit=True) as c:
        for t in TABLES:
            attributes=c.execute('SELECT attname,atttypid FROM pg_attribute WHERE attrelid=%s::regclass AND attnum>0 AND NOT attisdropped ORDER BY attnum',(t,)).fetchall()
            cols=[r[0] for r in attributes]
            rows=c.execute(sql.SQL('SELECT {} FROM {} ORDER BY id').format(sql.SQL(',').join(map(sql.Identifier,cols)),sql.Identifier(t))).fetchall()
            result[t]={'cols':cols,'oids':[r[1] for r in attributes],'rows':rows}
    return result

def logical_hash(rows):
    return hashlib.sha256(json.dumps(rows,sort_keys=True,default=str,separators=(',',':')).encode()).hexdigest()

    events=[]; state={'selector_serial':0,'claim_serial':0,'poll_serial':0}
    def stamp(event,**data):
        events.append({'event':event,'wall_ns':time.time_ns(),'mono_ns':time.perf_counter_ns(),
            'pid':os.getpid(),'backend_pid':state.get('backend_pid',backend),'role':role,'poll_serial':state['poll_serial'],
            'selector_serial':state['selector_serial'],'claim_serial':state['claim_serial'],**data})

    original_poll=worker.process_pending_conversations
    original_claim=worker._claim_turn
    original_run=worker._run_turn
    original_db_now=durable.db_now
    original_enter=transaction.Atomic.__enter__
    original_exit=transaction.Atomic.__exit__
    original_time=b9e_profile.time
    def poll(*a,**kw):
        state['poll_serial']+=1; stamp('poll_start')
        try: return original_poll(*a,**kw)
        finally: stamp('poll_end')
    def claim(conv,now=None):
        state['claim_serial']+=1; state['claim']=conv; state['claim_queries']=0
        stamp('claim_start',conversation_id=conv)
        result=None
        try:
            result=original_claim(conv,now)
            return result
        finally:
            stamp('claim_return',conversation_id=conv,turnwork_id=result.turn_work_id if result else None,
                query_count=state['claim_queries'])
            state.pop('claim',None)
    def run(witness):
        state['work']=witness.turn_work_id
        stamp('turn_start',turnwork_id=witness.turn_work_id)
        try: return original_run(witness)
        finally:
            stamp('turn_end',turnwork_id=witness.turn_work_id)
            state.pop('work',None)
    def db_now():
        value=original_db_now()
        if 'claim' in state: stamp('claim_db_clock',clock=value)
        return value
    def enter(atomic):
        outer=not connection.in_atomic_block
        result=original_enter(atomic)
        if outer and 'claim' in state: stamp('claim_transaction_enter')
        return result
    def leave(atomic,*a):
        outer=connection.in_atomic_block and not connection.savepoint_ids and 'claim' in state
        if outer: stamp('claim_commit_start')
        result=original_exit(atomic,*a)
        if outer: stamp('claim_commit_end',exception=bool(a[0]))
        return result
    def sql(execute,query,params,many,context):
        actual_backend=context['connection'].connection.info.backend_pid
        if actual_backend!=state.get('backend_pid',backend):
            old_backend=state.get('backend_pid',backend)
            state['backend_pid']=actual_backend
            stamp('backend_changed',previous_backend_pid=old_backend)
        selector='ROW_NUMBER()' in query and 'agent_conversation' in query
        is_claim='claim' in state
        if selector:
            state['selector_serial']+=1
            stamp('selector_start',sql=query,params=params)
        if is_claim:
            state['claim_queries']+=1
            stamp('claim_query_start',sql=query,locked=('FOR UPDATE' in query or 'FOR NO KEY UPDATE' in query or query.startswith('UPDATE')),
                skip_locked='SKIP LOCKED' in query)
        try: return execute(query,params,many,context)
        finally:
            if selector: stamp('selector_end')
            if is_claim: stamp('claim_query_end')
    class ClockProxy:
        def __getattr__(self,name): return getattr(original_time,name)
        def sleep(self,seconds):
            caller=sys._getframe(1).f_code.co_name
            event='poll_sleep' if caller=='worker' else 'fake_io_sleep'
            stamp(event+'_start',seconds=seconds,turnwork_id=state.get('work'))
            try: return original_time.sleep(seconds)
            finally: stamp(event+'_end',seconds=seconds,turnwork_id=state.get('work'))
    with ExitStack() as stack:
        for obj,attr,value in ((worker,'process_pending_conversations',poll),(module,'process_pending_conversations',poll),
            (worker,'_claim_turn',claim),(worker,'_run_turn',run),(durable,'db_now',db_now),
            (transaction.Atomic,'__enter__',enter),(transaction.Atomic,'__exit__',leave),(b9e_profile,'time',ClockProxy())):
            stack.enter_context(patch.object(obj,attr,value))
        stack.enter_context(connection.execute_wrapper(sql))
        try: yield
        finally:
            stamp('stop_clock')
            dump(OUT/('temporal-'+role+'.json.gz'),{'diagnostic_only':True,'role':role,'python_pid':os.getpid(),
                'backend_pid':backend,'clocks':{name:vars(time.get_clock_info(name)) for name in ('time','perf_counter')},'events':events})

        receipt.update(diagnostic_only=True,acceptance_authority=False,historical_profile_replaced=False,

            works=list(TurnWork.objects.filter(organization_id__in=a[3]).values('id','organization_id','conversation_id','status','claimed_at','completed_at','llm_calls_admitted')))

    def monitor():
        import psycopg
        while not stop_monitor.is_set():
            try:
                with connect(name,autocommit=True,row_factory=psycopg.rows.dict_row) as c:
                    rows=c.execute("SELECT pid,state,wait_event_type,wait_event,query_start FROM pg_stat_activity WHERE datname=current_database() AND pid<>pg_backend_pid()").fetchall()
                    locks=c.execute('SELECT pid,locktype,mode,granted FROM pg_locks WHERE NOT granted AND pid IN (SELECT pid FROM pg_stat_activity WHERE datname=current_database())').fetchall()
                    activities.append({'wall_ns':time.time_ns(),'activity':rows,'waiting_locks':locks})
                if len(activities)%40==0: snapshots.append(snapshot(name))
            except psycopg.OperationalError: pass
            stop_monitor.wait(.25)
    thread=threading.Thread(target=monitor); thread.start()
    try:
        print('TEMPORAL diagnostic full duration: warmup30 + observe300 + drain20; rates/topology frozen',flush=True)

    finally:
        stop_monitor.set(); thread.join()
        for reader in readers: reader.join(timeout=10)
        assert not any(reader.is_alive() for reader in readers)
    snapshots.append(snapshot(name))
    dump(OUT/'temporal-postgresql.json.gz',{'activity_samples':activities,'snapshots':snapshots,'historical':False})
    drop(name)
    print('TEMPORAL complete diagnostic only',flush=True)
