def read(name):
    p=OUT/name
    return json.load(gzip.open(p,'rt',encoding='utf-8')) if p.suffix=='.gz' else json.loads(p.read_text(encoding='utf-8'))
def save(name,obj):
    p=OUT/name
    assert not p.exists(),p
    with gzip.open(p,'wt',encoding='utf-8') as f: json.dump(obj,f,indent=2); f.write('\n')
def ns(text):
    value=datetime.fromisoformat(text)
    return int(value.timestamp())*1_000_000_000+value.microsecond*1000
def iso(value):
    return datetime.fromtimestamp(value/1e9,timezone.utc).isoformat(timespec='microseconds')
def stats(values):
    values=sorted(values)
    return {'n':len(values),'median':statistics.median(values),'p95':values[math.ceil(.95*len(values))-1],'max':max(values)} if values else None
def nodes(node):
    yield node
    for child in node.get('Plans',[]): yield from nodes(child)
def message_nodes(record):
    return [{k:n.get(k) for k in ('Node Type','Alias','Subplan Name','Actual Loops','Actual Rows','Plan Rows','Actual Total Time','Shared Hit Blocks','Shared Read Blocks')}
        for n in nodes(record['explain_event']['Plan']) if n.get('Relation Name')=='channels_message']
def pairs(events,start,end,key=None):
    active={}; result=[]
    for event in events:
        if event['event'] not in (start,end): continue
        serial=event.get(key) if key else None
        if event['event']==start:
            assert serial not in active
            active[serial]=event
        else:
            first=active.pop(serial)
            result.append((first,event))
    assert not active,(start,active)
    return result
def intersect(first,last,start,end): return max(0,min(last,end)-max(first,start))
