def utc():
    return datetime.now(timezone.utc).isoformat(timespec='microseconds')

def write(name, value):
    path = OUT / name
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, default=str) + '\n', encoding='utf-8')

def command(*args):
    return subprocess.run(['rtk', 'proxy', *map(str, args)], check=True,
                          stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
                          creationflags=subprocess.CREATE_NO_WINDOW)

def closed():
    with socket.socket() as sock:
        return sock.connect_ex(('127.0.0.1', PORT)) != 0

def historic():
    here = ROOT / 'docs/evals/profile-beta-1-e4-2026-10-01'
    diag = json.loads((here / 'base-diagnostics.json').read_text())
    selectors = [json.loads(line) for line in (here / 'sql-observations.jsonl').read_text().splitlines()]
    selectors = [r for r in selectors if r['event'] == 'selector_sql' and r['database'] == 'test_b8b_pb1e4_base']
    def stamp(s):
        return datetime.fromisoformat(s.replace(' UTC', '+00:00'))
    role_info, roles, events = {}, {}, {}
    for role in ('planner-1', 'planner-2'):
        entries = []
        for line in (here / f'base-{role}.log').read_text().splitlines():
            if line.startswith('B9E_READY '):
                role_info[role] = json.loads(line.split(' ', 1)[1])
            if not line.startswith('{'):
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError:
                continue
            if row.get('event') == 'AGENT_TURN_CLAIMED':
                roles[row['turn_work_id']] = role
            if 'ts' in row:
                # Formatter uses localtime but labels Z. A cross-source six-hour
                # offset is inferred; retained event precision remains one second.
                row['utc_second_inferred'] = stamp(row['ts']) + timedelta(hours=6)
                entries.append(row)
        events[role] = entries
    rows = [dict(m, org_index=o['org_index']) for o in diag['organizations'][:2] for m in o['messages']]
    rows.sort(key=lambda m: (stamp(m['claimed_at'])-stamp(m['created_at'])).total_seconds())
    assert len(rows) == 60
    rank_index = __import__('math').ceil(.95 * (len(rows)-1))
    assert rows[rank_index]['work_id'] == 86
    witnesses = []
    for rank in range(rank_index-2, len(rows)):
        m = rows[rank]
        created, eligible, claim = map(stamp, (m['created_at'], m['available_at'], m['claimed_at']))
        role = roles[m['work_id']]
        backend = role_info[role]['backend_pid']
        recent = [r for r in selectors if r['pid'] == backend and eligible <= stamp(r['timestamp']) <= claim]
        last = recent[-1]
        logtime = stamp(last['timestamp'])
        estimated_start = logtime - timedelta(milliseconds=last['milliseconds'])
        neighborhood = {
            key: [e for e in entries if eligible-timedelta(seconds=4) <= e['utc_second_inferred'] <= claim+timedelta(seconds=4)
                  and e['event'] in ('AGENT_WORKER_STARTED', 'AGENT_WORKER_STOPPED', 'AGENT_TURN_CLAIMED', 'AGENT_TURN_COMPLETED')]
            for key, entries in events.items()
        }
        all_backend_sql = [r for r in selectors if eligible-timedelta(seconds=4) <= stamp(r['timestamp']) <= claim+timedelta(seconds=2)]
        witnesses.append(dict(rank_one_based=rank+1, **m, role=role, python_pid=role_info[role]['pid'],
            backend_pid=backend, persisted_eligible_s=(eligible-created).total_seconds(),
            eligible_claim_s=(claim-eligible).total_seconds(), persisted_claim_s=(claim-created).total_seconds(),
            selector_records_same_backend_since_eligible=recent,
            last_selector_log_utc=logtime, last_selector_start_estimate_utc=estimated_start,
            pre_last_selector_estimate_s=(estimated_start-eligible).total_seconds(),
            post_last_selector_log_to_claim_s=(claim-logtime).total_seconds(),
            nearby_worker_events=neighborhood, nearby_selector_records=all_backend_sql))
    write('final-history.json', dict(head=SHA, historical=True, total_samples=60, rank_index_zero_based=rank_index,
        p95_work=86, p95_s=10.015949, workers=role_info, witnesses=witnesses,
        limitations=[
            'SQL observer dropped raw SQL and parameters; selector was identified by ROW_NUMBER and agent_conversation.',
            'Timestamp is PostgreSQL log emission with millisecond precision, not a captured execution-start/end pair.',
            'Start estimate subtracts logged duration from log timestamp; logging overhead and wall-clock error are not bounded.',
            'Worker timestamps have one-second resolution and localtime mislabeled Z; six-hour conversion is inferred.',
            'claimed_at is statement_timestamp at entry to _claim_turn_locked, before membership/CAS/commit.',
            'No historical event plan, complete per-claim query count, waits, pg_stats, pages or visibility snapshot survived.']))
    print(json.dumps(dict(historical_witnesses=len(witnesses), p95_work=86, workers=role_info)), flush=True)

class Trace:
    def __init__(self):
        self.rows = []
    def __call__(self, execute, sql, params, many, context):
        start_utc, start_ns = utc(), time.perf_counter_ns()
        try:
            return execute(sql, params, many, context)
        finally:
            end_ns, end_utc = time.perf_counter_ns(), utc()
            self.rows.append(dict(sql=sql, params=params, many=many, start_utc=start_utc,
                end_utc=end_utc, start_perf_ns=start_ns, end_perf_ns=end_ns,
                client_execute_ms=(end_ns-start_ns)/1e6))
